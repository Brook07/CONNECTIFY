app.post('/api/get-users-by-tags-and-distance', (req, res) => {
  const { userId, tags } = req.body;

  if (!userId) return res.status(400).json({ error: 'userId is required' });

  const getUserLocation = `SELECT latitude, longitude FROM users WHERE id = ? LIMIT 1;`;

  db.query(getUserLocation, [userId], (err, locResult) => {
    if (err || locResult.length === 0) {
      return res.status(500).json({ error: 'Failed to get user location' });
    }

    const { latitude, longitude } = locResult[0];

    let query = '';
    let params = [];

    if (tags && tags.length > 0) {
      const placeholders = tags.map(() => '?').join(',');
      query = `
        SELECT u.id AS userId, u.email, u.latitude, u.longitude,
          (CASE 
            WHEN u.latitude IS NOT NULL AND u.longitude IS NOT NULL THEN
              (6371 * acos(
                cos(radians(?)) * cos(radians(u.latitude)) *
                cos(radians(u.longitude) - radians(?)) +
                sin(radians(?)) * sin(radians(u.latitude))
              ))
            ELSE NULL
          END) AS distance
        FROM preferences p
        JOIN users u ON p.user_id = u.id
        WHERE p.preference IN (${placeholders}) AND u.id != ?
        GROUP BY u.id
        ORDER BY
          CASE WHEN distance IS NULL THEN 1 ELSE 0 END,
          distance ASC;
      `;
      params = [latitude, longitude, latitude, ...tags, userId];
    } else {
      query = `
        SELECT u.id AS userId, u.email, u.latitude, u.longitude,
          (CASE 
            WHEN u.latitude IS NOT NULL AND u.longitude IS NOT NULL THEN
              (6371 * acos(
                cos(radians(?)) * cos(radians(u.latitude)) *
                cos(radians(u.longitude) - radians(?)) +
                sin(radians(?)) * sin(radians(u.latitude))
              ))
            ELSE NULL
          END) AS distance
        FROM preferences p
        JOIN users u ON p.user_id = u.id
        WHERE p.preference IN (
          SELECT preference FROM preferences WHERE user_id = ?
        ) AND u.id != ?
        GROUP BY u.id
        ORDER BY
          CASE WHEN distance IS NULL THEN 1 ELSE 0 END,
          distance ASC;
      `;
      params = [latitude, longitude, latitude, userId, userId];
    }

    db.query(query, params, (err2, result) => {
      if (err2) {
        console.error('Error running combined query:', err2);
        return res.status(500).json({ error: 'Failed to retrieve users' });
      }

      if (result.length === 0) {
        return res.status(200).json({ message: 'No matching users found.', users: [] });
      }

      res.status(200).json({ message: 'Users filtered by tags and sorted by proximity', users: result });
    });
  });
});
